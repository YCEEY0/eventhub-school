    <script>
        function pointOnRect(top, left, bottom, right, x, y) {
            if (x > left && x < right && y > top && y < bottom) {
                return "inside";
            }

            if (x < left || x > right || y < top || y > bottom) {
                return "outside";
            }

            return "border";
        }

        function showResult() {
            const top = Number(document.getElementById("top").value);
            const left = Number(document.getElementById("left").value);
            const bottom = Number(document.getElementById("bottom").value);
            const right = Number(document.getElementById("right").value);
            const x = Number(document.getElementById("x").value);
            const y = Number(document.getElementById("y").value);

            const result = document.getElementById("result");

            if (top >= bottom || left >= right) {
                result.style.display = "block";
                result.style.background = "#fff0f0";
                result.style.color = "#c0392b";
                result.textContent =
                    "Invalid rectangle: top must be less than bottom, and left must be less than right.";
                drawRectangle();
                return;
            }

            const position = pointOnRect(top, left, bottom, right, x, y);

            result.style.display = "block";
            result.style.background =
                position === "inside" ? "#edf7f0" :
                position === "border" ? "#fff4df" : "#fff0f0";

            result.style.color =
                position === "inside" ? "#287545" :
                position === "border" ? "#946200" : "#c0392b";

            result.textContent = "Point is " + position + ".";

            drawRectangle();
        }

        function drawRectangle() {
            const canvas = document.getElementById("canvas");
            const ctx = canvas.getContext("2d");

            const top = Number(document.getElementById("top").value);
            const left = Number(document.getElementById("left").value);
            const bottom = Number(document.getElementById("bottom").value);
            const right = Number(document.getElementById("right").value);
            const x = Number(document.getElementById("x").value);
            const y = Number(document.getElementById("y").value);

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            if (top >= bottom || left >= right) {
                ctx.fillStyle = "#777";
                ctx.fillText("Enter valid rectangle coordinates.", 20, 30);
                return;
            }

            const padding = 35;
            const scale = Math.min(
                (canvas.width - padding * 2) / (right - left),
                (canvas.height - padding * 2) / (bottom - top)
            );

            const rectWidth = (right - left) * scale;
            const rectHeight = (bottom - top) * scale;
            const startX = (canvas.width - rectWidth) / 2;
            const startY = (canvas.height - rectHeight) / 2;

            ctx.fillStyle = "#e6f0ff";
            ctx.fillRect(startX, startY, rectWidth, rectHeight);

            ctx.strokeStyle = "#3d73b9";
            ctx.lineWidth = 2;
            ctx.strokeRect(startX, startY, rectWidth, rectHeight);

            ctx.fillStyle = "#333";
            ctx.font = "13px Arial";
            ctx.fillText("(" + left + ", " + top + ")", startX, startY - 10);
            ctx.fillText("(" + right + ", " + bottom + ")",
                startX + rectWidth - 90, startY + rectHeight + 18);

            const px = startX + (x - left) * scale;
            const py = startY + (y - top) * scale;
            const position = pointOnRect(top, left, bottom, right, x, y);

            ctx.beginPath();
            ctx.arc(px, py, 6, 0, Math.PI * 2);

            ctx.fillStyle =
                position === "inside" ? "#219653" :
                position === "border" ? "#e6a117" : "#d64545";

            ctx.fill();

            ctx.fillStyle = "#222";
            ctx.fillText("(" + x + ", " + y + ")", px + 10, py - 10);
        }

        showResult();
    </script>
